    providers.push(GoogleProvider({
      clientId: process.env.GOOGLE_CLIENT_ID,
      clientSecret: process.env.GOOGLE_CLIENT_SECRET

  return providers;
}

3.4 — CSV import endpoint with mapping (Step 5-7)
Replace the simple upload endpoint in doc_08 with this complete version:
// POST /api/csv-import
app.post("/api/csv-import", upload.single("file"), async (req, res) => {
  const entity = req.config.entities.find(e => e.name === req.body.entity);
  if (!entity) return res.status(400).json({ error: "ENTITY_NOT_FOUND" });

  const mapping = JSON.parse(req.body.mapping);
  // mapping = { csvCol: entityFieldId }

  const text = req.file.buffer.toString("utf-8");
  const { data } = Papa.parse(text, { header: true });

  const schema = buildZodSchema(entity);
  let imported = 0, skipped = 0;
  const errors = [];

  for (const [index, row] of data.entries()) {
    // Apply mapping: remap CSV columns to entity field names
    const mapped = {};
    for (const [csvCol, fieldId] of Object.entries(mapping)) {
      if (fieldId) mapped[fieldId] = row[csvCol];
    }

    const result = schema.safeParse(mapped);
    if (!result.success) {
      skipped++;
      errors.push({ row: index + 1, errors: result.error.errors });
      continue;
    }

    await db(entity.name).insert({
      app_id: req.app.id, user_id: req.user.id, data: result.data
    });
    imported++;
  }

  return res.json({ success: true, imported, skipped, errors: errors.slice(0, 10) });
});

Section 4 — Missing CRUD Handlers
doc_05 only documents Create and List handlers. Update and Delete are missing entirely. These are required for CRUD operations.

4.1 — Update handler
function updateHandler(entity) {
  const schema = buildZodSchema(entity);
  return withTenantScope(async (req, res) => {
    const { id } = req.params;
    const validation = schema.partial().safeParse(req.body.data);
    // .partial() makes all fields optional for PATCH-style updates
    if (!validation.success) {
      return res.status(400).json({ error: "VALIDATION_ERROR", details: validation.error.errors });
    }
    try {
      const existing = await db(entity.name)
        .where({ id, app_id: req.scope.app_id, user_id: req.scope.user_id })
        .first();
      if (!existing) return res.status(404).json({ error: "NOT_FOUND" });

      const merged = { ...existing.data, ...validation.data };
      const [updated] = await db(entity.name)
        .where({ id, app_id: req.scope.app_id, user_id: req.scope.user_id })
        .update({ data: merged, updated_at: new Date() })
        .returning("*");

      eventBus.emit("entity.update", { entity: entity.name, data: updated });
      return res.json({ success: true, data: updated });
    } catch (err) {
      return res.status(500).json({ error: "DB_ERROR", message: err.message });
    }
  });
}

4.2 — Delete handler
function deleteHandler(entity) {
  return withTenantScope(async (req, res) => {
    const { id } = req.params;
    try {
      const count = await db(entity.name)
        .where({ id, app_id: req.scope.app_id, user_id: req.scope.user_id })
        .delete();

      if (count === 0) return res.status(404).json({ error: "NOT_FOUND" });

      eventBus.emit("entity.delete", { entity: entity.name, id });
      return res.json({ success: true });
    } catch (err) {
      return res.status(500).json({ error: "DB_ERROR", message: err.message });
    }
  });
}

5.3 — POST /config
This is how a new config is loaded at runtime (used by the LLM generation flow and manual uploads):
app.post("/config", authMiddleware, express.json({ limit: "256kb" }), async (req, res) => {
  try {
    const result = validateConfig(req.body);
    if (!result.success) {
      return res.status(400).json({ error: "INVALID_CONFIG", errors: result.errors });
    }
    const newConfig = normalizeConfig(result.data);
    const changes = diffConfigs(runtimeState.config, newConfig);
    const hasBreaking = changes.some(c => isBreaking(c));
    if (hasBreaking) {
      return res.status(409).json({
        error: "BREAKING_SCHEMA_CHANGE",
        details: changes.filter(c => isBreaking(c))
      });
    }
    reloadConfig(newConfig);
    return res.json({ success: true, version: newConfig.version });
  } catch (err) {
    return res.status(500).json({ error: "RELOAD_FAILED", message: err.message });
  }
});

export function useRuntimeConfig() {
  const [config, setConfig] = useState(null);
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(true);  // add this

  useEffect(() => {
    fetch("/config/runtime")
      .then(res => {
        if (!res.ok) throw new Error("Config fetch failed: " + res.status);
        return res.json();
      })
      .then(data => { setConfig(data.config); setLoading(false); })
      .catch(err => { setError(err); setLoading(false); });
  }, []);

  return { config, error, loading };
}

6.3 — Form submission loading state
export function FormPage({ entity }) {
  const [formData, setFormData] = useState({});
  const [submitting, setSubmitting] = useState(false);
  const [submitError, setSubmitError] = useState(null);
  const [submitSuccess, setSubmitSuccess] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setSubmitting(true);
    setSubmitError(null);
    try {
      const res = await fetch(`/api/${entity.name}`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ data: formData })
      });
      if (!res.ok) {
        const json = await res.json();
        throw new Error(json.error ?? "Submission failed");
      }
      setSubmitSuccess(true);
    } catch (err) {
      setSubmitError(err.message);
    } finally {
      setSubmitting(false);
    }
  };

  if (submitSuccess) return <div style={{color:"green"}}>Saved successfully!</div>;

  return (
    <form onSubmit={handleSubmit}>
      {entity.fields.map(field => <div key={field.id}>{renderField({...field, ...})}</div>)}
      {submitError && <div style={{color:"red"}}>{submitError}</div>}
      <button type="submit" disabled={submitting}>
        {submitting ? "Saving..." : "Submit"}
      </button>
    </form>
  );
}

9.3 — Answers to hard judge questions
Q: How is this different from Retool?
Retool requires a visual editor. ConfigForge is code-first — the entire app is a JSON file you can version-control, diff, and generate with AI. You can also hot-reload configs without redeploying. Retool cannot do that.

Q: What happens if the config is wrong?
The validator rejects the config with a structured error object that shows the exact field path and what went wrong. The system never enters a partial state — it's fail-fast. The old config stays running until a valid one is provided.

Q: What haven't you built yet?
Honest answer: plugin marketplace, enterprise SSO, complex business logic workflows, and deep UI customization. ConfigForge is for standard CRUD applications. For highly custom logic, it currently requires backend extension code rather than pure config.

9.4 — Fallback plan if LLM fails during demo
Always have a pre-generated config file ready. If the Anthropic API call fails during the live demo:
Switch to the config editor tab
Paste your pre-prepared task-manager config JSON
Hit 'Load Config' manually
Say: 'The LLM pipeline is working — I just want to show you the rest of the system'
